cmake_minimum_required(VERSION 3.18)
project(lesson21_integrated_tensorrt_video_pipeline LANGUAGES CXX)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs)

add_library(integrated_pipeline_core src/pipeline_core.cpp)
add_library(integrated_frame_scheduler src/frame_scheduler.cpp)
target_include_directories(integrated_frame_scheduler PUBLIC include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(integrated_frame_scheduler PUBLIC integrated_pipeline_core ${OpenCV_LIBS})
target_compile_features(integrated_frame_scheduler PRIVATE cxx_std_17)
set_target_properties(integrated_frame_scheduler PROPERTIES CXX_EXTENSIONS OFF)
target_include_directories(integrated_pipeline_core PUBLIC include)
target_compile_features(integrated_pipeline_core PRIVATE cxx_std_17)
set_target_properties(integrated_pipeline_core PROPERTIES CXX_EXTENSIONS OFF)

add_executable(integrated_tensorrt_video_pipeline src/main.cpp)
target_link_libraries(integrated_tensorrt_video_pipeline PRIVATE integrated_pipeline_core)
target_compile_features(integrated_tensorrt_video_pipeline PRIVATE cxx_std_17)
set_target_properties(integrated_tensorrt_video_pipeline PROPERTIES CXX_EXTENSIONS OFF)

option(LESSON21_ENABLE_CUDA "Build CUDA/TensorRT backend" ON)
if(LESSON21_ENABLE_CUDA)
  enable_language(CUDA)
  find_library(NVINFER_LIB nvinfer REQUIRED)
  find_package(CUDAToolkit REQUIRED)
  add_library(integrated_tensorrt_backend src/tensorrt_backend.cu)
  target_include_directories(integrated_tensorrt_backend PUBLIC include ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(integrated_tensorrt_backend PUBLIC integrated_pipeline_core ${OpenCV_LIBS} ${NVINFER_LIB} CUDA::nppig CUDA::cudart)
  target_compile_features(integrated_tensorrt_backend PRIVATE cxx_std_17)
  set_target_properties(integrated_tensorrt_backend PROPERTIES CXX_EXTENSIONS OFF CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON CUDA_EXTENSIONS OFF)
endif()
if(LESSON21_ENABLE_CUDA)

if(BUILD_TESTING AND LESSON21_ENABLE_CUDA)
  set(LESSON21_TEST_ENGINE "${CMAKE_CURRENT_SOURCE_DIR}/../17_dynamic_batching/outputs/yolov8n_batch1_4_fp16.engine" CACHE FILEPATH "Dynamic TensorRT test engine")
  set(LESSON21_TEST_IMAGE "${CMAKE_CURRENT_SOURCE_DIR}/../assets/img.jpeg" CACHE FILEPATH "GPU test image")
  if(EXISTS "${LESSON21_TEST_ENGINE}" AND EXISTS "${LESSON21_TEST_IMAGE}")
    add_test(NAME integrated_gpu_batch1 COMMAND integrated_tensorrt_gpu_smoke "${LESSON21_TEST_ENGINE}" "${LESSON21_TEST_IMAGE}" 1)
    add_test(NAME integrated_gpu_batch4_two_slots COMMAND integrated_tensorrt_gpu_smoke "${LESSON21_TEST_ENGINE}" "${LESSON21_TEST_IMAGE}" 4 --two-slots)
    add_test(NAME integrated_gpu_reject_oversize COMMAND integrated_tensorrt_gpu_smoke "${LESSON21_TEST_ENGINE}" "${LESSON21_TEST_IMAGE}" 5)
    set_tests_properties(integrated_gpu_reject_oversize PROPERTIES WILL_FAIL TRUE)
  endif()

  add_test(NAME integrated_gpu_invalid_source COMMAND integrated_tensorrt_video_pipeline_gpu "${LESSON21_TEST_ENGINE}" /missing/image.jpg 8 4 2 "${LESSON21_TEST_OUTPUT}")
