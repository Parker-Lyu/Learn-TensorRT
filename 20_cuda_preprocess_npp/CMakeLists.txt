cmake_minimum_required(VERSION 3.18)
project(cuda_preprocess_npp LANGUAGES CXX CUDA)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CUDA_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 75 80 86 89)
endif()
include(CTest)
find_package(CUDAToolkit REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs)

add_library(cuda_preprocess src/preprocess.cpp src/cuda_preprocess.cu)
target_compile_features(cuda_preprocess PUBLIC cxx_std_17)
target_include_directories(cuda_preprocess PUBLIC include)
target_link_libraries(cuda_preprocess PUBLIC opencv_core opencv_imgproc CUDA::cudart CUDA::nppc CUDA::nppig CUDA::nvtx3)
set_target_properties(cuda_preprocess PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON CUDA_EXTENSIONS OFF)

add_executable(cuda_preprocess_npp src/main.cpp)
target_compile_features(cuda_preprocess_npp PRIVATE cxx_std_17)
target_link_libraries(cuda_preprocess_npp PRIVATE cuda_preprocess opencv_imgcodecs)

if(BUILD_TESTING)
    add_executable(cuda_preprocess_tests tests/test_preprocess.cpp)
    target_compile_features(cuda_preprocess_tests PRIVATE cxx_std_17)
    target_link_libraries(cuda_preprocess_tests PRIVATE cuda_preprocess)
    add_test(NAME cuda_preprocess_tests COMMAND cuda_preprocess_tests)
endif()
